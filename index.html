flags = transpose(flags_data)

// Dropdown input to select a country flag
viewof selectedFlag = Inputs.select(flags, {
  label: "Select a country:",
  format: d => d.name
})
pixelCache = new Map()

flagPixels = file => {
  if (!pixelCache.has(file)) {
    pixelCache.set(file, new Promise((resolve, reject) => {
      const image = new Image()
      image.onload = () => {
        const canvas = document.createElement("canvas")
        canvas.width = 32
        canvas.height = 20
        const context = canvas.getContext("2d")
        if (!context) {
          reject(new Error("Could not create a canvas to compare flags."))
          return
        }

        context.fillStyle = "#ffffff"
        context.fillRect(0, 0, canvas.width, canvas.height)
        const scale = Math.min(canvas.width / image.width, canvas.height / image.height)
        const width = image.width * scale
        const height = image.height * scale
        context.drawImage(image, (canvas.width - width) / 2, (canvas.height - height) / 2, width, height)
        resolve(context.getImageData(0, 0, canvas.width, canvas.height).data)
      }
      image.onerror = () => reject(new Error(`Could not load flag image: ${file}`))
      image.src = file
    }))
  }
  return pixelCache.get(file)
}

mostDifferentFlag = {
  const selectedPixels = await flagPixels(selectedFlag.file)
  const alternatives = await Promise.all(
    flags
      .filter(flag => flag.file !== selectedFlag.file)
      .map(async flag => {
        const pixels = await flagPixels(flag.file)
        let difference = 0
        for (let i = 0; i < selectedPixels.length; i += 4) {
          difference += Math.abs(selectedPixels[i] - pixels[i])
          difference += Math.abs(selectedPixels[i + 1] - pixels[i + 1])
          difference += Math.abs(selectedPixels[i + 2] - pixels[i + 2])
        }
        return { flag, difference }
      })
  )
  return alternatives.reduce((mostDifferent, current) =>
    current.difference > mostDifferent.difference ? current : mostDifferent
  ).flag
}
// Show the chosen flag alongside its closest visual contrast match.
html`
<div style="display: grid; grid-template-columns: repeat(auto-fit, minmax(240px, 1fr)); gap: 1.5rem;">
  <div>
    <h3>${selectedFlag.name}</h3>
    <img
      src="${selectedFlag.file}"
      alt="Flag of ${selectedFlag.name}"
      style="max-width: 500px; width: 100%; border: 1px solid #ccc; border-radius: 4px;"
    />
  </div>
  <div>
    <h3>Most different: ${mostDifferentFlag.name}</h3>
    <img
      src="${mostDifferentFlag.file}"
      alt="Flag of ${mostDifferentFlag.name}, the most visually different match"
      style="max-width: 500px; width: 100%; border: 1px solid #ccc; border-radius: 4px;"
    />
  </div>
</div>
`
